import { Meta, ColorPalette, ColorItem } from '@storybook/addon-docs/blocks';
import css from './tokens.css?raw';

<Meta title="Foundations/Tokens" />

export const tokens = [...css.matchAll(/(--[\w-]+):\s*([^;]+);/g)].map(([, name, value]) => ({
  name,
  value: value.trim(),
  // --color-primary-500 -> color/primary/500, --font-size-sm -> font-size/sm
  figma: name.startsWith('--color-')
    ? name.slice(2).replaceAll('-', '/')
    : name.slice(2).replace(/-([^-]+)$/, '/$1'),
}));
export const group = (prefix) => tokens.filter((t) => t.name.startsWith(prefix));
export const colorFamilies = [...new Set(group('--color-').map((t) => t.name.split('-')[3]))];

export const cell = { padding: '8px 12px', borderBottom: '1px solid #eee', textAlign: 'left', verticalAlign: 'middle' };
export const code = { fontFamily: 'ui-monospace, monospace', fontSize: 13 };

export const TokenTable = ({ prefix, preview }) => (
  <table style={{ width: '100%', borderCollapse: 'collapse' }}>
    <thead>
      <tr>
        <th style={cell}>Preview</th>
        <th style={cell}>CSS variable</th>
        <th style={cell}>Figma variable</th>
        <th style={cell}>Value</th>
      </tr>
    </thead>
    <tbody>
      {group(prefix).map((t) => (
        <tr key={t.name}>
          <td style={cell}>{preview(t)}</td>
          <td style={{ ...cell, ...code }}>{`var(${t.name})`}</td>
          <td style={{ ...cell, ...code }}>{t.figma}</td>
          <td style={{ ...cell, ...code }}>{t.value}</td>
        </tr>
      ))}
    </tbody>
  </table>
);

## Text styles

export const textStyles = [...new Set(group('--text-').map((t) => t.name.match(/^--text-(.+?)-(font|line|letter)/)[1]))];
export const capitalize = (s) => s[0].toUpperCase() + s.slice(1);

<table style={{ width: '100%', borderCollapse: 'collapse' }}>
  <thead>
    <tr>
      <th style={cell}>Preview</th>
      <th style={cell}>CSS class</th>
      <th style={cell}>Figma text style</th>
      <th style={cell}>Properties</th>
    </tr>
  </thead>
  <tbody>
    {textStyles.map((name) => (
      <tr key={name}>
        <td style={cell}>
          {/* Inline font-family: Storybook's docs typography otherwise overrides the class here. */}
          <span className={`text-${name}`} style={{ fontFamily: `var(--text-${name}-font-family)` }}>What are we analysing today..?</span>
        </td>
        <td style={{ ...cell, ...code }}>{`.text-${name}`}</td>
        <td style={{ ...cell, ...code }}>{capitalize(name)}</td>
        <td style={{ ...cell, ...code, whiteSpace: 'pre' }}>
          {group(`--text-${name}-`).map((t) => `${t.name.replace(`--text-${name}-`, '')}: ${t.value}`).join('\n')}
        </td>
      </tr>
    ))}
  </tbody>
</table>
